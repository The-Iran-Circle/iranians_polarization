ابتدا یک حوزه و سپس یک سؤال را انتخاب کنید

هفت حوزهٔ موضوعی پیمایش، شامل ده‌ها سؤال، در این کاوشگر در دسترس‌اند: از ایده‌های سیاسی پایه تا گرایش‌های اقتدارگرایانه، بنیادهای اخلاقی، نگرش‌های سیاست‌گذاری، تعریف ایرانی بودن و نمادهای ملی. برای هر سؤال، نمودار توزیع پاسخ‌ها بر حسب گروه سیاسی، میانگین و فاصلهٔ اطمینان ۹۵٪ و اندازهٔ شکاف میان گروه‌ها (Cohen’s d) را می‌بینید.

labels    = FileAttachment("data/labels.json").json()
responses = FileAttachment("data/responses.json").json()
summary   = FileAttachment("data/summary.json").json()
statsAll  = FileAttachment("data/stats.json").json()

// Battery options for dropdown 1
batteryOptions = Object.entries(labels.batteries).map(([id, b]) => ({
  id, label_fa: b.label_fa, label_en: b.label_en
}))

groupOrder = labels.groupings.pol_group.levels.map(l => l.id)
groupLabelMap = Object.fromEntries(
  labels.groupings.pol_group.levels.map(l => [l.id, l.label_fa])
)
groupColors = ["#7d3c98", "#16a085", "#2980b9", "#d68910", "#c0392b"]
viewof selectedBattery = Inputs.select(
  batteryOptions.map(o => o.id),
  {
    label: "حوزه:",
    format: id => batteryOptions.find(o => o.id === id).label_fa,
    value: "Symbolic_nation"
  }
)
batteryDef  = labels.batteries[selectedBattery]
itemOptions = batteryDef.items
viewof selectedItem = Inputs.select(
  itemOptions.map(o => o.id),
  {
    label: "سؤال:",
    format: id => {
      const it = itemOptions.find(o => o.id === id);
      return it ? it.label_fa : id;
    }
  }
)
itemMeta = itemOptions.find(o => o.id === selectedItem) ?? itemOptions[0]
scale    = batteryDef.scale
rowResponses = responses.filter(r => r.item === selectedItem)
rowSummary   = summary.filter(s => s.item === selectedItem)
itemStats    = statsAll.find(s => s.item === selectedItem)
rowSummarySorted = groupOrder
  .map(g => rowSummary.find(s => s.group === g))
  .filter(Boolean)

// Scale-aware ticks: 1..N at integer positions, with Persian labels
xTicks = Array.from({length: scale.max - scale.min + 1}, (_, i) => scale.min + i)
xTickFormat = v => {
  const idx = v - scale.min;
  return scale.ticks_fa[idx] || String(v);
}
// ─── The plot ───────────────────────────────────────────────────────────
Plot.plot({
  width: 760,
  height: 340,
  marginLeft: 180,
  marginRight: 30,
  marginTop: 30,
  marginBottom: 70,
x: {
    domain: [scale.min - 1.5, scale.max + 0.5],
    label: null,
    ticks: xTicks,
    tickFormat: xTickFormat
  },
  y: {
    domain: groupOrder.filter(g => rowSummarySorted.some(s => s.group === g)),
    label: null
  },
  color: {
    domain: groupOrder,
    range: groupColors,
    legend: false
  },
  marks: [
    // Midpoint reference
    Plot.ruleX([scale.midpoint], { stroke: "#bbb", strokeDasharray: "3,3" }),

    // Faded jittered points — using PRE-COMPUTED offsets (jx, jy)
    // so each respondent's point lands in a stable, distinct spot.
    Plot.dot(rowResponses, {
      x: d => d.value + d.jx,
      y: "group",
      dy: d => d.jy * 18,             // y is categorical; convert jy (~±0.32) to pixels
      fill: "group",
      r: 2.6,
      fillOpacity: 0.28,
      stroke: "none"
    }),

    // 95% CI bars
    Plot.ruleY(rowSummarySorted, {
      x1: "ci_lo", x2: "ci_hi", y: "group",
      stroke: "group", strokeWidth: 2.5
    }),

    // Mean dots
    Plot.dot(rowSummarySorted, {
      x: "mean", y: "group", fill: "group",
      stroke: "white", strokeWidth: 1.5, r: 7
    }),

    // n labels at right edge of plot
    Plot.text(rowSummarySorted, {
      x: scale.max + 0.45, y: "group",
      text: d => `n = ${d.n}`,
      textAnchor: "end", fill: "#777", fontSize: 10
    }),

    // Y-axis: Persian group labels
    Plot.axisY({ tickFormat: g => groupLabelMap[g] || g, fontSize: 12 }),

    // X-axis title in Persian
    Plot.axisX({ anchor: "bottom", label: null })
  ]
})
function formatP(p) {
  if (p < 0.001) return "p < 0.001";
  return `p = ${p.toFixed(3)}`;
}

sizeFa = ({ negligible: "ناچیز", small: "کوچک", medium: "متوسط", large: "بزرگ" })

meaningfulPairs = (itemStats?.pairwise || [])
  .filter(p => p.d != null && Math.abs(p.d) >= 0.2)
  .sort((a, b) => Math.abs(b.d) - Math.abs(a.d))

groupsPresent = groupOrder.filter(g => rowSummarySorted.some(s => s.group === g)).length
df1 = groupsPresent - 1
df2 = (itemStats?.n_total ?? 0) - groupsPresent

html`
  <div class="stats-readout">
    <div class="item-question">${itemMeta.full_fa || itemMeta.label_fa}</div>
    <div class="item-meta">
      <strong>${itemMeta.label_fa}</strong>
      <span style="color:#888">— ${itemMeta.label_en}</span>
    </div>

    <strong>تحلیل واریانس:</strong>
    <span class="num">F(${df1}, ${df2}) = ${itemStats.F.toFixed(2)}</span>,
    <span class="num">${formatP(itemStats.p)}</span>,
    <span class="num">η² = ${itemStats.eta_squared.toFixed(3)}</span>
    <br>

    <strong>قابل‌توجه‌ترین شکاف‌ها میان گروه‌ها</strong> (Cohen's d):<br>
    ${meaningfulPairs.length === 0
      ? '<span style="color:#888">هیچ شکاف معناداری میان گروه‌ها مشاهده نشد.</span>'
      : meaningfulPairs.slice(0, 5).map(p => {
          const g1 = groupLabelMap[p.group1] || p.group1;
          const g2 = groupLabelMap[p.group2] || p.group2;
          const d  = Math.abs(p.d).toFixed(2);
          return `· <span class="effect-${p.size}">${g1} ↔ ${g2}: ` +
                 `<span class="num">d = ${d}</span> (${sizeFa[p.size]})</span>`;
        }).join("<br>")
    }
  </div>
`

نکات مهم در تفسیر

  • اندازهٔ اثر مهم‌تر از معناداری آماری است. d بین ۰٫۲ تا ۰٫۵ شکاف کوچکی است؛ بین ۰٫۵ تا ۰٫۸ متوسط؛ بالاتر از ۰٫۸ بزرگ.
  • مقیاس‌ها متفاوت‌اند. بعضی سؤال‌ها روی مقیاس ۷ نقطه‌ای موافقت پرسیده شده‌اند (کاملاً مخالف → کاملاً موافق)، برخی روی ۵ نقطه‌ای اهمیت، و برخی روی ۵ نقطه‌ای شدت. به محور افقی هر نمودار دقت کنید.
  • آیتم‌های با پسوند «معکوس» در مقیاس‌های شخصیتی (SDO، ACT، MFQ) برای کنترل سوگیری پاسخ طراحی شده‌اند؛ پاسخ بالا در آن‌ها به معنای کم بودن سازهٔ مدنظر است. در نمرهٔ نهایی مقیاس باید معکوس شوند، اما در این کاوشگر پاسخ‌های خام را می‌بینید.
  • نقاط کم‌رنگ پاسخ‌های فردی‌اند تا توزیع واقعی هم دیده شود، نه فقط میانگین.
